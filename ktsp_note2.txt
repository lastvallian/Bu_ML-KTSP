1. 输出整个层次树结构（Tree Structure）

例如：

Root: split using K-TSP pairs [(12, 45), (67, 123)]
 ├── Left child (class 5)
 └── Right child split using pairs [(7, 88)]
      ├── class 2
      └── class 8


Hierarchical Classifier 本质是一个分类树，你需要看每个节点如何二分类。

我可以帮你写一个很漂亮的 print_tree()：

def print_tree(self):
    def dfs(node, depth=0):
        prefix = "  " * depth
        if node.label is not None:
            print(f"{prefix}Leaf → Class {node.label}")
        else:
            print(f"{prefix}Node → K-TSP pairs: {node.pairs}")
        dfs(node.left, depth+1)
        dfs(node.right, depth+1)
    dfs(self.root)

2.2. 输出每个节点的准确率（Node-level performance）

你可以查看：哪个节点最难分类？

添加：

def evaluate_nodes(self, X, y):
    results = []

    def dfs(node, samples):
        if node.label is not None:
            # leaf
            correct = np.sum(y[samples] == node.label)
            total = len(samples)
            acc = correct / total if total > 0 else 0
            results.append((node.label, acc, total))
            return
        
        # internal node
        p = node.classifier.predict(X[:, samples])
        left_samples = samples[p == 1]
        right_samples = samples[p == 0]

        dfs(node.left, left_samples)
        dfs(node.right, right_samples)

    dfs(self.root, np.arange(X.shape[1]))
    return results

node_acc = hclf.evaluate_nodes(X, y)
for label, acc, n in node_acc:
    print(f"Class {label}: samples={n}, leaf accuracy={acc:.4f}")


4. 输出特征对的重要性（Pair Importance）

K-TSP 的 score 已经可以表示重要性。

你可以加：

def get_all_pairs(self):
    pairs = []

    def dfs(node):
        if node and node.pairs:
            pairs.extend(node.pairs)
        if node.left:
            dfs(node.left)
        if node.right:
            dfs(node.right)

    dfs(self.root)
    return pairs


调用：

print("All K-TSP pairs used in the whole tree:")
print(hclf.get_all_pairs())

def print_tree(node, indent=""):
    if node.label is not None:
        print(indent + f"Leaf → Class: {node.label}")
        return
    
    print(indent + f"Node: K-TSP pairs = {node.pairs}")
    print(indent + " Left:")
    print_tree(node.left, indent + "   ")
    print(indent + " Right:")
    print_tree(node.right, indent + "   ")

import matplotlib.pyplot as plt

def plot_pair(X, y, pair):
    i, j = pair
    plt.figure(figsize=(6,5))
    plt.scatter(X[i], X[j], c=y)
    plt.xlabel(f"Feature {i}")
    plt.ylabel(f"Feature {j}")
    plt.title(f"K-TSP Pair: ({i}, {j})")
    plt.show()

plot_pair(X, y, hclf.get_all_pairs()[0][0])

def pair_accuracy(X, y, pair):
    i, j = pair
    pred = (X[i] > X[j]).astype(int)
    # 真实标签需要二值版
    y_bin = (y == y[0]).astype(int)
    return np.mean(pred == y_bin)

for p in hclf.get_all_pairs()[0]:
    print(p, pair_accuracy(X, y, p))
